cmake_minimum_required(VERSION 3.21)
project(Rolly VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(ROLLY_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(ROLLY_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LibArchive REQUIRED IMPORTED_TARGET libarchive)

add_library(rolly_core src/core/theme.cpp src/core/theme_source.cpp src/core/storage.cpp src/core/archive.cpp)
target_include_directories(rolly_core PUBLIC src)
target_link_libraries(rolly_core PUBLIC Qt6::Core Qt6::Gui PkgConfig::LibArchive)

add_library(rolly_ui src/ui/icons.cpp src/ui/archive_model.cpp src/ui/job.cpp src/ui/file_picker.cpp
    src/ui/modal.cpp src/ui/modal_motion.cpp src/ui/theme_watcher.cpp src/ui/window.cpp)
target_link_libraries(rolly_ui PUBLIC rolly_core Qt6::Widgets Qt6::DBus)

add_executable(rolly src/main.cpp src/cli.cpp)
target_link_libraries(rolly PRIVATE rolly_ui)
foreach(target rolly_core rolly_ui rolly)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    foreach(name theme archive ui)
        add_executable(${name}_tests tests/${name}_tests.cpp)
        target_link_libraries(${name}_tests PRIVATE rolly_ui Qt6::Test)
        add_test(NAME ${name} COMMAND ${name}_tests)
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endforeach()
    add_test(NAME cli COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh $<TARGET_FILE:rolly>)
endif()

include(GNUInstallDirs)
install(TARGETS rolly RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.rolly.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.rolly.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
